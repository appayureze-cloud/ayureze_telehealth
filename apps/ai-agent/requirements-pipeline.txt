# Day 6: real-time multilingual translation pipeline.
#
# Installed separately from requirements.txt (the Day 5 agent lifecycle
# has no dependency on these) because this pulls in ~2GB of ML packages
# and model weights. Install with:
#   pip install -r requirements.txt
#   pip install -r requirements-pipeline.txt --index-url https://download.pytorch.org/whl/cpu --extra-index-url https://pypi.org/simple
#
# torch is pinned to the CPU-only wheel index deliberately — the default
# PyPI torch wheel pulls in the full CUDA/NVIDIA toolkit (~15GB) even on a
# machine with no GPU, which is wasted space/bandwidth for this build's
# CPU-only inference. See docs/ai/README.md "Known limitations".

onnxruntime==1.20.1      # Silero VAD (ONNX, no torch needed for VAD itself)
faster-whisper==1.1.1    # STT (CTranslate2-based, no torch needed)
langid==1.1.6            # language identification (pure Python, no ML runtime)

torch==2.14.0+cpu        # required by transformers (translation + TTS models below)
# transformers bumped 2026-10-01: Trivy's container scan (running for real
# against the ai-agent image for the first time, see
# .github/workflows/security.yml) found 4 HIGH-severity deserialization/RCE
# CVEs in 4.47.1 (CVE-2024-11392/-11393/-11394, CVE-2026-4372/-5241/-9856),
# whose highest "Fixed Version" floor was 5.10.0. 5.10.1 is the oldest
# release at/above every floor (5.10.0 itself isn't published on PyPI) —
# picked over the latest (5.18.0) to keep this a targeted bump. Re-verified
# against this repo's full ai-agent test suite (232/232, same pass count as
# before) and that every transformers class this codebase actually imports
# (AutoModelForSeq2SeqLM, AutoTokenizer, MarianMTModel, MarianTokenizer,
# M2M100ForConditionalGeneration, M2M100Tokenizer, T5ForConditionalGeneration,
# T5Tokenizer, VitsModel) still imports cleanly under 5.10.1.
